cmake_minimum_required(VERSION 3.16)
project(squeakview_yolo26_pose_parser LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/DeepStream.cmake")
set(CUDA_INCLUDES "/usr/local/cuda/include" CACHE PATH "CUDA headers; no CUDA code is built")
add_library(nvdsinfer_squeakview_pose SHARED parser.cpp)
set_target_properties(nvdsinfer_squeakview_pose PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_features(nvdsinfer_squeakview_pose PRIVATE cxx_std_17)
target_include_directories(nvdsinfer_squeakview_pose SYSTEM PRIVATE "${DEEPSTREAM_INCLUDES}" "${CUDA_INCLUDES}")
# Preserve baseline letterbox rounding in optimized builds (avoid fused multiply/add).
target_compile_options(nvdsinfer_squeakview_pose PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_options(nvdsinfer_squeakview_pose PRIVATE -Wl,--no-undefined)
include(CTest)
if(BUILD_TESTING)
  add_executable(pose_parser_boundary ../nvdsinfer_custom_impl_yolo/tests/test_pose_parser.cpp)
  target_compile_features(pose_parser_boundary PRIVATE cxx_std_17)
  target_include_directories(pose_parser_boundary SYSTEM PRIVATE "${DEEPSTREAM_INCLUDES}" "${CUDA_INCLUDES}")
  target_compile_definitions(pose_parser_boundary PRIVATE SQUEAKVIEW_POSE_PARSER_SOURCE="${CMAKE_CURRENT_LIST_DIR}/parser.cpp")
  target_compile_options(pose_parser_boundary PRIVATE -UNDEBUG -ffp-contract=off)
  add_test(NAME pose_parser_boundary COMMAND pose_parser_boundary)
  add_executable(compare_pose_parsers tests/compare.cpp)
  target_compile_features(compare_pose_parsers PRIVATE cxx_std_17)
  target_include_directories(compare_pose_parsers SYSTEM PRIVATE "${DEEPSTREAM_INCLUDES}" "${CUDA_INCLUDES}")
  target_link_libraries(compare_pose_parsers PRIVATE ${CMAKE_DL_LIBS})
  set(BASELINE_PARSER "" CACHE FILEPATH "Existing legacy or packaged parser for comparison")
  if(BASELINE_PARSER)
    add_test(NAME pose_parser_equivalence COMMAND compare_pose_parsers "${BASELINE_PARSER}" $<TARGET_FILE:nvdsinfer_squeakview_pose>)
  endif()
endif()
